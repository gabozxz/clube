-- TransporteEscolar v3 — Supabase Setup
create table if not exists escolas (id bigserial primary key, nome text not null, nivel text, tipo text, "end" text, mun text, tel text, dir text, status text default 'Ativa', obs text, created_at timestamptz default now());
create table if not exists motoristas (id bigserial primary key, nome text not null, cpf text, tel text, cnh text, cat text, val date, nasc date, status text default 'Ativo', obs text, created_at timestamptz default now());
create table if not exists veiculos (id bigserial primary key, placa text not null unique, tipo text, marca text, modelo text not null, ano int, cap int, cor text, renavam text, crlv date, status text default 'Disponível', obs text, created_at timestamptz default now());
create table if not exists linhas (id bigserial primary key, codigo text not null, nome text not null, motorista_id bigint references motoristas(id) on delete set null, veiculo_id bigint references veiculos(id) on delete set null, turno text, status text default 'Ativa', rotas jsonb default '[]', aluno_ids bigint[] default '{}', vagas int default 30, obs text, created_at timestamptz default now());
create table if not exists alunos (id bigserial primary key, nome text not null, nasc date, sexo text, serie text, turno text, resp text, tel text, "end" text, linha_id bigint references linhas(id) on delete set null, escola_id bigint references escolas(id) on delete set null, status text default 'Ativo', obs text, created_at timestamptz default now());
alter table escolas disable row level security;
alter table motoristas disable row level security;
alter table veiculos disable row level security;
alter table linhas disable row level security;
alter table alunos disable row level security;
